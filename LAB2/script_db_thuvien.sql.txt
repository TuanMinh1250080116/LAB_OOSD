-- Trigger: Tự động cập nhật số bản khả dụng khi thêm/bớt bản sao sách vật lý
CREATE OR ALTER TRIGGER TRG_CapNhatSoBanSao
ON BanSaoSach
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    
    -- Cập nhật lại cho các đầu sách bị ảnh hưởng
    UPDATE ds
    SET 
        TongSoBan = (SELECT COUNT(*) FROM BanSaoSach b WHERE b.MaDauSach = ds.MaDauSach AND b.TrangThai != N'Thanh lý'),
        SoBanKhaDung = (SELECT COUNT(*) FROM BanSaoSach b WHERE b.MaDauSach = ds.MaDauSach AND b.TrangThai = N'Sẵn sàng')
    FROM DauSach ds
    WHERE ds.MaDauSach IN (
        SELECT DISTINCT MaDauSach FROM inserted
        UNION
        SELECT DISTINCT MaDauSach FROM deleted
    );
END;
GO

-- Procedure 1: Đăng ký mượn sách giữ chỗ (UC05 - Đăng ký mượn sách)
CREATE OR ALTER PROCEDURE sp_DangKyMuonSach
    @MaThe VARCHAR(20),
    @MaDauSach INT,
    @MaDangKy INT OUTPUT,
    @ThongBao NVARCHAR(255) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    BEGIN TRY
        -- 1. Kiểm tra hiệu lực thẻ thư viện
        IF NOT EXISTS (SELECT 1 FROM TheThuVien WHERE MaThe = @MaThe AND TrangThai = N'Hoạt động' AND NgayHetHan >= CAST(SYSDATETIME() AS DATE))
        BEGIN
            SET @ThongBao = N'Mã thẻ thư viện không tồn tại hoặc đã hết hiệu lực!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- 2. Kiểm tra độc giả có sách quá hạn không
        IF EXISTS (
            SELECT 1 
            FROM PhieuMuon pm
            JOIN ChiTietPhieuMuon ct ON pm.MaPhieuMuon = ct.MaPhieuMuon
            WHERE pm.MaThe = @MaThe 
              AND ct.TrangThai = N'Đang mượn' 
              AND ct.HanTra < CAST(SYSDATETIME() AS DATE)
        )
        BEGIN
            SET @ThongBao = N'Độc giả đang có sách quá hạn chưa trả, không thể đăng ký mượn tiếp!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- 3. Kiểm tra số sách đang mượn + đang giữ chỗ với hạn mức cho phép
        DECLARE @SoLuongHienTai INT, @SoLuongToiDa INT;
        SELECT @SoLuongToiDa = SoSachToiDa FROM TheThuVien WHERE MaThe = @MaThe;
        
        SELECT @SoLuongHienTai = (
            SELECT COUNT(*) FROM ChiTietPhieuMuon ct 
            JOIN PhieuMuon pm ON ct.MaPhieuMuon = pm.MaPhieuMuon 
            WHERE pm.MaThe = @MaThe AND ct.TrangThai = N'Đang mượn'
        ) + (
            SELECT COUNT(*) FROM DangKyMuon dk 
            WHERE dk.MaThe = @MaThe AND dk.TrangThai = N'Chờ nhận sách'
        );

        IF (@SoLuongHienTai >= @SoLuongToiDa)
        BEGIN
            SET @ThongBao = N'Độc giả đã đạt số lượng sách tối đa được phép mượn!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- 4. Kiểm tra số bản khả dụng và lấy 1 cuốn có trạng thái 'Sẵn sàng'
        DECLARE @MaBanSaoChon VARCHAR(30);
        SELECT TOP 1 @MaBanSaoChon = MaBanSao 
        FROM BanSaoSach WITH (UPDLOCK, ROWLOCK)
        WHERE MaDauSach = @MaDauSach AND TrangThai = N'Sẵn sàng';

        IF @MaBanSaoChon IS NULL
        BEGIN
            SET @ThongBao = N'Đầu sách này hiện tại đã hết bản in khả dụng!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- 5. Cập nhật bản sao sang 'Đã giữ chỗ'
        UPDATE BanSaoSach 
        SET TrangThai = N'Đã giữ chỗ' 
        WHERE MaBanSao = @MaBanSaoChon;

        UPDATE DauSach 
        SET SoBanKhaDung = SoBanKhaDung - 1 
        WHERE MaDauSach = @MaDauSach;

        -- 6. Tạo đơn đăng ký (Hạn nhận sách: 48 tiếng)
        INSERT INTO DangKyMuon (MaThe, MaDauSach, MaBanSao, NgayDangKy, HanNhanSach, TrangThai)
        VALUES (@MaThe, @MaDauSach, @MaBanSaoChon, SYSDATETIME(), DATEADD(HOUR, 48, SYSDATETIME()), N'Chờ nhận sách');

        SET @MaDangKy = SCOPE_IDENTITY();
        SET @ThongBao = N'Đăng ký giữ chỗ thành công!';
        
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SET @ThongBao = ERROR_MESSAGE();
    END CATCH
END;
GO

-- Procedure 2: Lấy danh sách cần gửi email nhắc nhở trước hạn 3 ngày (UC17 - Gửi email nhắc hạn trả)
CREATE OR ALTER PROCEDURE sp_LayDanhSachCanNhacHanTra
AS
BEGIN
    SET NOCOUNT ON;

    -- Lọc các phiếu có hạn trả đúng 3 ngày tới và chưa được nhắc
    SELECT 
        ct.MaChiTiet,
        pm.MaPhieuMuon,
        tk.HoTen AS TenDocGia,
        tk.Email AS EmailDocGia,
        ds.TuaDe AS TenSach,
        ct.MaBanSao,
        ct.HanTra
    FROM ChiTietPhieuMuon ct
    JOIN PhieuMuon pm ON ct.MaPhieuMuon = pm.MaPhieuMuon
    JOIN TheThuVien tv ON pm.MaThe = tv.MaThe
    JOIN TaiKhoan tk ON tv.MaTaiKhoan = tk.MaTaiKhoan
    JOIN BanSaoSach bss ON ct.MaBanSao = bss.MaBanSao
    JOIN DauSach ds ON bss.MaDauSach = ds.MaDauSach
    WHERE ct.TrangThai = N'Đang mượn'
      AND ct.DaNhacEmail = 0
      AND DATEDIFF(DAY, CAST(SYSDATETIME() AS DATE), ct.HanTra) = 3
      AND tk.Email IS NOT NULL 
      AND tk.Email LIKE '%_@__%.__%';
END;
GO

-- Procedure 3: Ghi nhận trả sách và tự động tính tiền phạt quá hạn (UC12 - Ghi nhận trả sách)
CREATE OR ALTER PROCEDURE sp_GhiNhanTraSach
    @MaChiTietPM INT,
    @MaThuThuNhan INT,
    @TinhTrangKhiTra NVARCHAR(100),
    @TienPhatHuHong DECIMAL(18,2) = 0,
    @KetQua NVARCHAR(255) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRANSACTION;

    BEGIN TRY
        DECLARE @MaBanSao VARCHAR(30), @MaDauSach INT, @HanTra DATE, @NgayTra DATE;
        SET @NgayTra = CAST(SYSDATETIME() AS DATE);

        SELECT 
            @MaBanSao = ct.MaBanSao, 
            @MaDauSach = bss.MaDauSach,
            @HanTra = ct.HanTra
        FROM ChiTietPhieuMuon ct
        JOIN BanSaoSach bss ON ct.MaBanSao = bss.MaBanSao
        WHERE ct.MaChiTiet = @MaChiTietPM AND ct.TrangThai = N'Đang mượn';

        IF @MaBanSao IS NULL
        BEGIN
            SET @KetQua = N'Không tìm thấy chi tiết mượn hoặc sách đã được trả!';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Tính số ngày quá hạn và tiền phạt (ví dụ: 5.000 VNĐ / ngày trễ)
        DECLARE @SoNgayTre INT = DATEDIFF(DAY, @HanTra, @NgayTra);
        DECLARE @TienPhatTre DECIMAL(18,2) = 0;

        IF @SoNgayTre > 0
            SET @TienPhatTre = @SoNgayTre * 5000;

        -- Cập nhật chi tiết phiếu mượn
        UPDATE ChiTietPhieuMuon
        SET 
            NgayTraThucTe = SYSDATETIME(),
            TinhTrangKhiTra = @TinhTrangKhiTra,
            TienPhatQuaHan = @TienPhatTre,
            TienPhatHuHong = @TienPhatHuHong,
            MaThuThuNhanSach = @MaThuThuNhan,
            TrangThai = N'Đã trả'
        WHERE MaChiTiet = @MaChiTietPM;

        -- Cập nhật lại bản sao sách
        UPDATE BanSaoSach
        SET 
            TrangThai = CASE WHEN @TinhTrangKhiTra = N'Hỏng' THEN N'Thanh lý' ELSE N'Sẵn sàng' END,
            TinhTrang = @TinhTrangKhiTra
        WHERE MaBanSao = @MaBanSao;

        -- Cập nhật lại số bản khả dụng của đầu sách
        IF @TinhTrangKhiTra != N'Hỏng'
        BEGIN
            UPDATE DauSach
            SET SoBanKhaDung = SoBanKhaDung + 1
            WHERE MaDauSach = @MaDauSach;
        END

        SET @KetQua = N'Tiếp nhận trả sách thành công! Tiền phạt trễ hạn: ' + CAST(@TienPhatTre AS NVARCHAR(20)) + N' VNĐ.';
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        SET @KetQua = ERROR_MESSAGE();
    END CATCH
END;
GO
